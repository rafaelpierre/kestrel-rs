    https://www.geeksforgeeks.org/python/python-taskgroups-with-asyncio/

    When using asyncio.TaskGroup with asyncio.run(), an unhandled exception in any task can cause the error "unhandled errors in a taskgroup (1 sub-exception)". This happens because Python 3.11+ wraps multiple task errors into an ExceptionGroup, which requires special handling with except*.

    Example:

    import asyncio

    async def worker(name, delay):
    await asyncio.sleep(delay)
    if name == "B":
    raise ValueError(f"{name} failed!")
    return f"{name} done"

    async def main():
    try:
    async with asyncio.TaskGroup() as tg:
    tg.create_task(worker("A", 1))
    tg.create_task(worker("B", 0.5))
    tg.create_task(worker("C", 2))
    except* ValueError as eg:
    print("Caught ValueErrors:")
    for e in eg.exceptions:
    print(f" - {e}")
    except* Exception as eg:
    print("Other exceptions:", eg.exceptions)

    asyncio.run(main())
